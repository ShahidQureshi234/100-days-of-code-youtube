Day 96 — AsyncIO.py
import asyncio
import time

async def fetch_data(id: int):
    print(f"Fetching {id}...")
    await asyncio.sleep(1)
    return {"id": id, "data": f"Result {id}"}

async def main():
    # Run 3 fetches concurrently
    results = await asyncio.gather(
        fetch_data(1),
        fetch_data(2),
        fetch_data(3)
    )
    print(f"Done: {len(results)} results in 1s")

# Instead of 3s sequential → 1s parallel
asyncio.run(main())
